## Introduction to Code Quality

Code quality refers to the attributes of code that make it easy to understand, maintain, and extend. High-quality code is essential for the long-term success of any frontend project, as it reduces technical debt, improves developer productivity, and enhances the overall user experience.

## Clean Code Principles

### Readability

Readable code is easy to understand at a glance, which makes it easier to maintain and debug.

#### Naming Conventions

Use clear, descriptive names for variables, functions, and classes.

```javascript
// Poor naming
const x = 86400000;
function fn(a, b) {
  return a * b;
}

// Good naming
const MILLISECONDS_IN_A_DAY = 86400000;
function calculateArea(width, height) {
  return width * height;
}
```

#### Consistent Formatting

Maintain consistent formatting throughout your codebase.

```javascript
// Inconsistent formatting
function calculateTotal(items){
    let total = 0;
  for(let i=0;i<items.length;i++)
  {
    total += items[i].price;
  }
    return total;
}

// Consistent formatting
function calculateTotal(items) {
  let total = 0;
  
  for (let i = 0; i < items.length; i++) {
    total += items[i].price;
  }
  
  return total;
}
```

### Function Design

#### Single Responsibility Principle

Each function should do one thing and do it well.

```javascript
// Function doing too much
function processUserData(userData) {
  // Validate user data
  if (!userData.name || !userData.email) {
    throw new Error('Invalid user data');
  }
  
  // Format user data
  const formattedData = {
    fullName: `${userData.name.first} ${userData.name.last}`,
    email: userData.email.toLowerCase(),
    // More formatting
  };
  
  // Save to database
  database.save('users', formattedData);
  
  // Send welcome email
  emailService.send({
    to: formattedData.email,
    subject: 'Welcome!',
    body: `Hello ${formattedData.fullName}!`
  });
  
  return formattedData;
}

// Better approach: separate concerns
function validateUserData(userData) {
  if (!userData.name || !userData.email) {
    throw new Error('Invalid user data');
  }
  return userData;
}

function formatUserData(userData) {
  return {
    fullName: `${userData.name.first} ${userData.name.last}`,
    email: userData.email.toLowerCase(),
    // More formatting
  };
}

function saveUserData(formattedData) {
  return database.save('users', formattedData);
}

function sendWelcomeEmail(formattedData) {
  return emailService.send({
    to: formattedData.email,
    subject: 'Welcome!',
    body: `Hello ${formattedData.fullName}!`
  });
}

// Orchestration function
function processUserData(userData) {
  const validatedData = validateUserData(userData);
  const formattedData = formatUserData(validatedData);
  saveUserData(formattedData);
  sendWelcomeEmail(formattedData);
  return formattedData;
}
```

#### Function Size

Keep functions small and focused.

```javascript
// Long, complex function

  }

    html += '<span class="verified-badge">✓</span>';

  html += `<li><span class="stat-count">${user.following}</span> Following</li>`;

  html += '</div>';
  
  // Render user posts
  html += '<div class="profile-posts">';
  html += '<h3>Recent Posts</h3>';
  if (user.posts && user.posts.length > 0) {
    html += '<ul class="post-list">';
    for (const post of user.posts) {
      html += `<li class="post-item">`;

// Better approach: smaller, focused functions
function renderProfileHeader(user) {

  html += user.avatar
    ? `<img src="${user.avatar}" alt="${user.name}" class="profile-avatar">`
    : `<div class="profile-avatar-placeholder">${user.name.charAt(0)}</div>`;
  html += `<h2 class="profile-name">${user.name}</h2>`;
  if (user.verified) {
    html += '<span class="verified-badge">✓</span>';
  }
  html += '</div>';
  return html;
}

function renderProfileDetails(user) {
  let html = '<div class="profile-details">';
  if (user.bio) {
    html += `<p class="profile-bio">${user.bio}</p>`;
  }
  html += '<ul class="profile-stats">';
  html += `<li><span class="stat-count">${user.followers}</span> Followers</li>`;
  html += `<li><span class="stat-count">${user.following}</span> Following</li>`;
  html += '</ul>';
  html += '</div>';
  return html;
}

function renderProfilePosts(posts) {
  let html = '<div class="profile-posts">';
  html += '<h3>Recent Posts</h3>';
  
  if (posts && posts.length > 0) {
    html += '<ul class="post-list">';
    for (const post of posts) {
      html += renderPostItem(post);
    }
    html += '</ul>';
  } else {
    html += '<p class="no-posts">No posts yet.</p>';
  }
  
  html += '</div>';
  return html;
}

function renderPostItem(post) {
  let html = '<li class="post-item">';
  html += `<h4>${post.title}</h4>`;
  html += `<p>${post.excerpt}</p>`;
  html += `<span class="post-date">${formatDate(post.date)}</span>`;
  html += '</li>';
  return html;
}

function renderUserProfile(user) {
  return (
    '<div class="profile">' +
    renderProfileHeader(user) +
    renderProfileDetails(user) +
    renderProfilePosts(user.posts) +
    '</div>'
  );
}
```

#### Pure Functions

When possible, write pure functions that don't have side effects.

```javascript
// Function with side effects
let total = 0;
function addToTotal(value) {
  total += value; // Side effect: modifies external state
  return total;
}

// Pure function
function add(a, b) {
  return a + b; // No side effects, same output for same input
}
```

### Code Organization

#### DRY (Don't Repeat Yourself)

Avoid duplicating code by extracting common functionality.

```javascript
// Repetitive code
function validateEmail(email) {
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailRegex.test(email)) {
    throw new Error('Invalid email format');
  }
  return true;
}

function validateUsername(username) {
  const usernameRegex = /^[a-zA-Z0-9_]{3,16}$/;
  if (!usernameRegex.test(username)) {
    throw new Error('Invalid username format');
  }
  return true;
}

function validatePassword(password) {
  const passwordRegex = /^(?=.*[A-Za-z])(?=.*\d)[A-Za-z\d]{8,}$/;
  if (!passwordRegex.test(password)) {
    throw new Error('Invalid password format');
  }
  return true;
}

// DRY approach
function validate(value, regex, errorMessage) {
  if (!regex.test(value)) {
    throw new Error(errorMessage);
  }
  return true;
}

function validateEmail(email) {
  return validate(
    email,
    /^[^\s@]+@[^\s@]+\.[^\s@]+$/,
    'Invalid email format'
  );
}

function validateUsername(username) {
  return validate(
    username,
    /^[a-zA-Z0-9_]{3,16}$/,
    'Invalid username format'
  );
}

function validatePassword(password) {
  return validate(
    password,
    /^(?=.*[A-Za-z])(?=.*\d)[A-Za-z\d]{8,}$/,
    'Invalid password format'
  );
}
```

#### KISS (Keep It Simple, Stupid)

Prefer simple solutions over complex ones.

```javascript
// Overly complex
function isEven(num) {
  return Boolean(!(num & 1));
}

// Simpler and more readable
function isEven(num) {
  return num % 2 === 0;
}
```

## Code Reviews

### Code Review Process

Establish a structured code review process to improve code quality.

#### Code Review Checklist

1. **Functionality**: Does the code work as expected?
2. **Readability**: Is the code easy to understand?
3. **Maintainability**: Will the code be easy to modify in the future?
4. **Performance**: Are there any performance concerns?
5. **Security**: Are there any security vulnerabilities?
6. **Error Handling**: Does the code handle errors appropriately?
7. **Testing**: Is the code adequately tested?
8. **Documentation**: Is the code properly documented?

#### Code Review Best Practices

- Focus on the code, not the person
- Be specific and constructive in feedback
- Use questions rather than statements when possible
- Provide examples or alternatives when suggesting changes
- Acknowledge good practices and improvements

## Refactoring

### When to Refactor

Refactoring should be a regular part of the development process.

#### Signs That Code Needs Refactoring

- Duplicate code
- Long functions or methods
- Large classes or components
- Deep nesting
- Unclear naming
- High coupling between modules
- Difficulty in adding new features
- Frequent bugs in a specific area

### Refactoring Techniques

#### Extract Function/Component

Break down large functions or components into smaller, more focused ones.

```javascript
// Before refactoring
function renderUserProfile(user) {
  let html = '<div class="profile">';
  
  // Render user header
  html += '<div class="profile-header">';
  // ... 20 more lines of header rendering code
  
  // Render user details
  html += '<div class="profile-details">';
  // ... 30 more lines of details rendering code
  
  html += '</div>';
  return html;
}

// After refactoring
function renderUserProfile(user) {
  return (
    '<div class="profile">' +
    renderProfileHeader(user) +
    renderProfileDetails(user) +
    '</div>'
  );
}

function renderProfileHeader(user) {
  // Implementation
}

function renderProfileDetails(user) {
  // Implementation
}
```

#### Rename Variable/Function

Improve naming to better reflect purpose.

```javascript
// Before refactoring
function fn(a, b) {
  return a * b;
}

// After refactoring
function calculateArea(width, height) {
  return width * height;
}
```

#### Replace Conditional with Polymorphism

Replace complex conditionals with polymorphic objects.

```javascript
// Before refactoring
function calculatePrice(product, quantity) {
  let basePrice = product.price * quantity;
  
  if (product.type === 'book') {
    return basePrice * 0.9; // 10% discount for books
  } else if (product.type === 'electronics') {
    return basePrice * 0.95; // 5% discount for electronics
  } else if (product.type === 'clothing') {
    return quantity > 2 ? basePrice * 0.8 : basePrice; // Buy 3+ clothing items, get 20% off
  } else {
    return basePrice;
  }
}

// After refactoring
class Product {
  constructor(price) {
    this.price = price;
  }
  
  calculatePrice(quantity) {
    return this.price * quantity;
  }
}

class Book extends Product {
  calculatePrice(quantity) {
    return super.calculatePrice(quantity) * 0.9; // 10% discount
  }
}

class Electronics extends Product {
  calculatePrice(quantity) {
    return super.calculatePrice(quantity) * 0.95; // 5% discount
  }
}

class Clothing extends Product {
  calculatePrice(quantity) {
    const basePrice = super.calculatePrice(quantity);
    return quantity > 2 ? basePrice * 0.8 : basePrice; // Buy 3+ items, get 20% off
  }
}

// Usage
function createProduct(type, price) {
  switch (type) {
    case 'book':
      return new Book(price);
    case 'electronics':
      return new Electronics(price);
    case 'clothing':
      return new Clothing(price);
    default:
      return new Product(price);
  }
}

function calculatePrice(product, quantity) {
  return product.calculatePrice(quantity);
}
```

### Safe Refactoring

Refactor code safely to avoid introducing bugs.

#### Refactoring with Tests

1. Ensure you have tests covering the code you're refactoring
2. Make small, incremental changes
3. Run tests after each change
4. Commit frequently

```javascript
// Test for the function we want to refactor
describe('calculatePrice', () => {
  test('applies 10% discount for books', () => {
    const book = { type: 'book', price: 20 };
    expect(calculatePrice(book, 2)).toBe(36); // 20 * 2 * 0.9 = 36
  });
  
  test('applies 5% discount for electronics', () => {
    const electronics = { type: 'electronics', price: 100 };
    expect(calculatePrice(electronics, 1)).toBe(95); // 100 * 1 * 0.95 = 95
  });
  
  test('applies 20% discount for 3+ clothing items', () => {
    const clothing = { type: 'clothing', price: 30 };
    expect(calculatePrice(clothing, 3)).toBe(72); // 30 * 3 * 0.8 = 72
  });
  
  test('applies no discount for other products', () => {
    const other = { type: 'other', price: 10 };
    expect(calculatePrice(other, 2)).toBe(20); // 10 * 2 = 20
  });
});

// After refactoring, the same tests should pass with the new implementation
describe('calculatePrice with polymorphism', () => {
  test('applies 10% discount for books', () => {
    const book = createProduct('book', 20);
    expect(calculatePrice(book, 2)).toBe(36);
  });
  
  // ... other tests
});
```

## Technical Debt Management

### Identifying Technical Debt

Recognize signs of technical debt in your codebase.

#### Common Types of Technical Debt

1. **Code Debt**: Poorly written or structured code
2. **Architecture Debt**: Suboptimal architectural decisions
3. **Test Debt**: Inadequate or missing tests
4. **Documentation Debt**: Outdated or missing documentation
5. **Dependency Debt**: Outdated or problematic dependencies

### Strategies for Managing Technical Debt

#### Technical Debt Inventory

Maintain an inventory of technical debt items.


#### Boy Scout Rule

Leave the code better than you found it.

```javascript
// When fixing a bug or adding a feature, also improve the code quality

// Original code with a bug
function calculateTotal(items) {
  var total = 0;
  for (var i = 0; i < items.length; i++) {
    total += items[i].price;
  }
  return total;
}

// Fixed bug and improved code quality
function calculateTotal(items) {
  if (!Array.isArray(items)) {
    throw new Error('Items must be an array');
  }
  
  return items.reduce((total, item) => {
    // Fixed bug: Check if item has price property
    if (!item || typeof item.price !== 'number') {
      return total;
    }
    return total + item.price;
  }, 0);
}
```

Allocate time specifically for refactoring.

- Dedicate a percentage of each sprint to technical debt reduction
- Schedule regular "refactoring days" or "tech debt sprints"
- Include refactoring tasks in the backlog with proper prioritization

## Documentation

### Code Documentation

Document your code to make it easier to understand and maintain.

#### JSDoc

Use JSDoc to document JavaScript code.

```javascript
/**
 * Calculates the total price of items in a shopping cart
 * @param {Array} items - Array of items with price property
 * @param {Object} options - Calculation options
 * @param {boolean} options.includeDiscounts - Whether to include discounts
 * @param {number} options.taxRate - Tax rate to apply (decimal)
 * @returns {number} The total price including tax
 * @throws {Error} If items is not an array
 * @example
 * const items = [
 *   { name: 'Product 1', price: 10 },
 *   { name: 'Product 2', price: 20 }
 * ];
 * const total = calculateTotal(items, { includeDiscounts: true, taxRate: 0.1 });
 * // Returns: 33 (30 + 10% tax)
 */
function calculateTotal(items, options = {}) {
  if (!Array.isArray(items)) {
    throw new Error('Items must be an array');
  }
  
  const { includeDiscounts = false, taxRate = 0 } = options;
  
  let subtotal = items.reduce((total, item) => total + item.price, 0);
  
  if (includeDiscounts) {
    subtotal = applyDiscounts(subtotal, items);
  }
  
  const tax = subtotal * taxRate;
  
  return subtotal + tax;
}
```

### Code Complexity Analysis

Measure and manage code complexity.

#### Cyclomatic Complexity

Cyclomatic complexity measures the number of linearly independent paths through a program's source code.

```javascript
// High cyclomatic complexity (bad)
function getDiscount(user, cart, coupon) {
  let discount = 0;
  
  if (user.isPremium) {
    discount += 0.1; // 10% discount for premium users
    
    if (cart.total > 100) {
      discount += 0.05; // Additional 5% for orders over $100
    }
    
    if (user.loyaltyPoints > 1000) {
      discount += 0.03; // Additional 3% for loyal customers
    }
  } else {
    if (cart.total > 200) {
      discount += 0.05; // 5% discount for orders over $200
    }
    
    if (user.firstPurchase) {
      discount += 0.1; // 10% discount for first purchase
    }
  }
  
  if (coupon) {
    if (coupon.type === 'percentage') {
      discount += coupon.value / 100;
    } else if (coupon.type === 'fixed') {
      discount += coupon.value / cart.total;
    }
    
    if (coupon.isStackable === false) {
      discount = Math.max(discount, coupon.value / 100);
    }
  }
  
  return Math.min(discount, 0.5); // Maximum 50% discount
}

// Lower cyclomatic complexity (better)
function getPremiumUserDiscount(user, cart) {
  let discount = 0.1; // 10% base discount
  
  if (cart.total > 100) {
    discount += 0.05;
  }
  
  if (user.loyaltyPoints > 1000) {
    discount += 0.03;
  }
  
  return discount;
}

function getRegularUserDiscount(user, cart) {
  let discount = 0;
  
  if (cart.total > 200) {
    discount += 0.05;
  }
  
  if (user.firstPurchase) {
    discount += 0.1;
  }
  
  return discount;
}

function getCouponDiscount(coupon, cart, currentDiscount) {
  if (!coupon) return currentDiscount;
  
  let couponDiscount = 0;
  
  if (coupon.type === 'percentage') {
    couponDiscount = coupon.value / 100;
  } else if (coupon.type === 'fixed') {
    couponDiscount = coupon.value / cart.total;
  }
  
  if (coupon.isStackable === false) {
    return Math.max(currentDiscount, couponDiscount);
  }
  
  return currentDiscount + couponDiscount;
}

function getDiscount(user, cart, coupon) {
  let discount = user.isPremium
    ? getPremiumUserDiscount(user, cart)
    : getRegularUserDiscount(user, cart);
  
  discount = getCouponDiscount(coupon, cart, discount);
  
  return Math.min(discount, 0.5); // Maximum 50% discount
}
```
